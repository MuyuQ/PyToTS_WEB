## 场景与问题

装饰器允许在不修改原始代码的情况下增强功能。Python 和 TypeScript 都支持装饰器，但实现和用法有差异。TypeScript 的装饰器是实验性特性，需要启用编译器选项。

**学完能做什么**

- 为方法编写计时、重试、缓存等通用装饰器
- 使用装饰器工厂传递配置参数（如重试次数、缓存大小）
- 实现类装饰器实现单例、只读属性等模式
- 配合 `reflect-metadata` 实现参数验证、依赖注入等元编程
- 区分装饰器执行顺序（属性/方法从上到下，类从下到上）
- 避免返回 `boolean` 而非类型谓词导致收窄失效

## Python 回顾

```python
import functools
from typing import Callable, Any

# 函数装饰器
def timer(func: Callable) -> Callable:
    @functools.wraps(func)
    def wrapper(*args, **kwargs) -> Any:
        import time
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} took {elapsed:.2f} seconds")
        return result
    return wrapper

@timer
def slow_function():
    import time
    time.sleep(1)
    return "Done"

# 带参数的装饰器
def retry(max_attempts: int = 3):
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"Attempt {attempt + 1} failed, retrying...")
            return wrapper
    return decorator

@retry(max_attempts=5)
def unstable_function():
    import random
    if random.random() < 0.7:
        raise Exception("Random failure")
    return "Success"

# 类装饰器
def singleton(cls):
    instances = {}
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return get_instance

@singleton
class Database:
    def __init__(self):
        self.connection = "Connected"

# 内置装饰器
class Person:
    def __init__(self, name: str):
        self._name = name

    @property
    def name(self) -> str:
        return self._name

    @name.setter
    def name(self, value: str) -> None:
        if not value:
            raise ValueError("Name cannot be empty")
        self._name = value

    @staticmethod
    def is_adult(age: int) -> bool:
        return age >= 18

    @classmethod
    def create_anonymous(cls) -> "Person":
        return cls("Anonymous")
```

```typescript
function retry(maxAttempts = 3) {
  return function(target, key, descriptor) {
    const original = descriptor.value;
    descriptor.value = async function(...args) {
      for (let i = 0; i < maxAttempts; i++) {
        try { return await original.apply(this, args); }
        catch (e) { if (i === maxAttempts-1) throw e; }
      }
    };
    return descriptor;
  };
}
```

```typescript
function singleton<T extends new (...args: any[]) => {}>(ctor: T) {
  let instance: InstanceType<T> | undefined;
  return class extends ctor {
    constructor(...args: any[]) {
      if (instance) return instance;
      super(...args);
      // 派生类实例与基类实例类型结构兼容，TS 无法自动推导，需显式断言
      instance = this as InstanceType<T>;
    }
  };
}
```

## TypeScript 等价写法

```typescript
// 启用装饰器需要在 tsconfig.json 中设置：
// "experimentalDecorators": true
// "emitDecoratorMetadata": true

// 方法装饰器
function timer(target: Object, propertyKey: string, descriptor: PropertyDescriptor) {
  const originalMethod = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const start = performance.now();
    const result = originalMethod.apply(this, args);
    const elapsed = performance.now() - start;
    console.log(`${propertyKey} took ${elapsed.toFixed(2)} ms`);
    return result;
  };

  return descriptor;
}

class Example {
  @timer
  slowMethod() {
    // 模拟耗时操作
    for (let i = 0; i < 1000000; i++) {}
    return "Done";
  }
}

// 装饰器工厂（带参数）
function retry(maxAttempts: number = 3) {
  return function (target: Object, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;

    descriptor.value = async function (...args: any[]) {
      for (let attempt = 0; attempt < maxAttempts; attempt++) {
        try {
          return await originalMethod.apply(this, args);
        } catch (error) {
          if (attempt === maxAttempts - 1) {
            throw error;
          }
          console.log(`Attempt ${attempt + 1} failed, retrying...`);
        }
      }
    };

    return descriptor;
  };
}

class ApiClient {
  @retry(5)
  async unstableRequest() {
    if (Math.random() < 0.7) {
      throw new Error("Random failure");
    }
    return "Success";
  }
}

// 类装饰器
function singleton<T extends new (...args: any[]) => {}>(constructor: T) {
  let instance: InstanceType<T> | undefined;

  return class extends constructor {
    constructor(...args: any[]) {
      if (instance) {
        return instance;
      }
      super(...args);
      // 派生类实例与基类实例类型结构兼容，TS 无法自动推导，需显式断言
      instance = this as InstanceType<T>;
    }
  };
}

@singleton
class Database {
  connection: string = "Connected";
}

// 属性装饰器：给静态配置字段加只读保护。
// 装饰器在类定义时就执行：静态字段先初始化、后冻结，顺序正好；
// 实例字段的初始化发生在实例化时（装饰之后），直接冻结会抛 TypeError。
function readonly(target: Object, propertyKey: string) {
  Object.defineProperty(target, propertyKey, {
    writable: false,
  });
}

class Config {
  @readonly
  static apiUrl: string = "https://api.example.com";
}

// 读取正常；重新赋值会抛 TypeError
console.log(Config.apiUrl);
```

```python
# Python 需手动管理缓存字典
def memoize(func):
    cache = {}
    def wrapper(*args):
        if args in cache: return cache[args]
        cache[args] = func(*args)
        return cache[args]
```

```typescript
function memoize<T extends (...args: any[]) => any>(
  target: Object, key: string, descriptor: TypedPropertyDescriptor<T>
) {
  const cache = new Map<string, ReturnType<T>>();
  descriptor.value = function(...args: Parameters<T>) {
    const k = JSON.stringify(args);
    if (cache.has(k)) return cache.get(k)!;
    const r = descriptor.value!.apply(this, args);
    cache.set(k, r); return r;
  } as T;
  return descriptor;
}
```

## 差异与常见陷阱

### 1. 装饰器执行顺序

```typescript
// TypeScript 装饰器执行顺序：
// 1. 属性/方法装饰器（从上到下）
// 2. 类装饰器（从下到上）

@ClassDecorator
class MyClass {
  @PropertyDecorator
  myProperty: string;

  @MethodDecorator
  myMethod() {}
}

// 执行顺序：
// 1. PropertyDecorator
// 2. MethodDecorator
// 3. ClassDecorator
```

:::tip
记忆口诀：成员从上到下，类从下到上。先装饰内部成员，最后包装整个类。
:::

### 2. 元数据反射

```typescript
// 需要安装 reflect-metadata
import "reflect-metadata";

const REQUIRED_KEY = Symbol("required");

// 参数装饰器 + 元数据
function required(target: Object, propertyKey: string | symbol, parameterIndex: number) {
  const existingRequired: number[] = Reflect.getMetadata(REQUIRED_KEY, target, propertyKey) || [];
  existingRequired.push(parameterIndex);
  Reflect.defineMetadata(REQUIRED_KEY, existingRequired, target, propertyKey);
}

// 方法装饰器读取元数据
function validate(target: Object, propertyKey: string, descriptor: PropertyDescriptor) {
  const method = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const requiredParameters: number[] =
      Reflect.getMetadata(REQUIRED_KEY, target, propertyKey) || [];

    for (const index of requiredParameters) {
      if (args[index] === undefined || args[index] === null) {
        throw new Error(`Missing required argument at position ${index}`);
      }
    }

    return method.apply(this, args);
  };
}

class UserService {
  @validate
  createUser(@required name: string, @required email: string, age?: number) {
    return { name, email, age };
  }
}
```

:::note
`reflect-metadata` 必须在所有装饰器之前导入（`import "reflect-metadata"`），且 `tsconfig.json` 需开启 `emitDecoratorMetadata: true`。遗漏任一项会导致 `Reflect.getMetadata` 返回 `undefined`。
:::

错误版忘记导入 reflect-metadata，正确版在入口文件最早导入：

```typescript
// ❌ 错误：未导入 reflect-metadata，元数据为 undefined
function required(target: Object, key: string | symbol, index: number) {
  Reflect.defineMetadata("required", [index], target, key);
}
@validate
class Service {
  create(@required name: string) {} // 运行时 requiredParameters 为 []
}

// ✅ 正确：入口文件第一行导入
import "reflect-metadata";
function required(target: Object, key: string | symbol, index: number) {
  const existing = Reflect.getMetadata("required", target, key) || [];
  existing.push(index);
  Reflect.defineMetadata("required", existing, target, key);
}
```

### 3. 类型安全的装饰器

```typescript
// 使用泛型让装饰器更类型安全
function memoize<T extends (...args: any[]) => any>(
  target: Object,
  propertyKey: string,
  descriptor: TypedPropertyDescriptor<T>
) {
  const originalMethod = descriptor.value!;
  const cache = new Map<string, ReturnType<T>>();

  descriptor.value = function (...args: Parameters<T>): ReturnType<T> {
    const key = JSON.stringify(args);

    if (cache.has(key)) {
      return cache.get(key)!;
    }

    const result = originalMethod.apply(this, args);
    cache.set(key, result);
    return result;
  } as T;

  return descriptor;
}

class Calculator {
  @memoize
  fibonacci(n: number): number {
    if (n <= 1) return n;
    return this.fibonacci(n - 1) + this.fibonacci(n - 2);
  }
}
```

:::tip
泛型装饰器用 `TypedPropertyDescriptor<T>` 保留方法签名，`as T` 断言让 TypeScript 信任修改后的描述符。若不加 `as T`，返回类型会变为 `any`。
:::

错误版未使用 TypedPropertyDescriptor 导致类型丢失，正确版泛型约束保留签名：

```typescript
// ❌ 错误：descriptor.value 类型变为 any，调用方失去类型提示
function badMemoize(target: Object, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function(...args: any[]) { return original.apply(this, args); };
  return descriptor;
}

// ✅ 正确：泛型约束保留参数与返回类型
function goodMemoize<T extends (...args: any[]) => any>(
  target: Object, key: string, descriptor: TypedPropertyDescriptor<T>
) {
  const original = descriptor.value!;
  const cache = new Map<string, ReturnType<T>>();
  descriptor.value = function(...args: Parameters<T>): ReturnType<T> {
    const k = JSON.stringify(args);
    if (cache.has(k)) return cache.get(k)!;
    const r = original.apply(this, args);
    cache.set(k, r); return r;
  } as T;
  return descriptor;
}
```

## 优势对比

| 特性       | Python            | TypeScript         |
| ---------- | ----------------- | ------------------ |
| 语法       | `@decorator`      | `@decorator`       |
| 类型安全   | 运行时            | 编译时             |
| 装饰器类型 | 函数/类/方法/属性 | 类/方法/属性/参数  |
| 元数据     | 手动管理          | `reflect-metadata` |
| 状态       | 正式特性          | 实验性特性         |
| 性能       | 运行时开销        | 编译后无开销       |

## 练习

将以下 Python 装饰器转换为 TypeScript：

```python
import functools
from typing import Callable, Any

def cache_result(max_size: int = 128):
    def decorator(func: Callable) -> Callable:
        cache = {}

        @functools.wraps(func)
        def wrapper(*args) -> Any:
            if args in cache:
                return cache[args]

            result = func(*args)

            if len(cache) >= max_size:
                cache.pop(next(iter(cache)))

            cache[args] = result
            return result

        return wrapper
    return decorator

@cache_result(max_size=100)
def expensive_calculation(n: int) -> int:
    print(f"Calculating for {n}")
    return n * n
```

```typescript
function cacheResult(maxSize: number = 128) {
  return function <T extends (...args: any[]) => any>(
    target: Object,
    propertyKey: string,
    descriptor: TypedPropertyDescriptor<T>
  ) {
    const originalMethod = descriptor.value!;
    const cache = new Map<string, ReturnType<T>>();

    descriptor.value = function (...args: Parameters<T>): ReturnType<T> {
      const key = JSON.stringify(args);

      if (cache.has(key)) {
        return cache.get(key)!;
      }

      const result = originalMethod.apply(this, args);

      if (cache.size >= maxSize) {
        const firstKey = cache.keys().next().value;
        cache.delete(firstKey);
      }

      cache.set(key, result);
      return result;
    } as T;

    return descriptor;
  };
}

class Calculator {
  @cacheResult(100)
  expensiveCalculation(n: number): number {
    console.log(`Calculating for ${n}`);
    return n * n;
  }
}
```

## 面试追问

1. **TypeScript 装饰器为什么是实验性的？**
   - 答：ECMAScript 装饰器提案还在发展中，TypeScript 的实现可能与最终标准不兼容。

2. **装饰器和继承的区别？**
   - 答：装饰器在不修改原类的情况下增强功能，继承创建新类。装饰器更灵活但增加复杂性。

3. **什么时候应该避免使用装饰器？**
   - 答：当简单函数调用就能满足需求时，或者团队不熟悉装饰器模式时。
