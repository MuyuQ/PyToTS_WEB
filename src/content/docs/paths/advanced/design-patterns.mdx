## 场景与问题

设计模式是解决常见软件设计问题的方案。Python 和 TypeScript 都可以实现这些模式，但 TypeScript 的类型系统让实现更明确。

**学完能做什么**

- 用 `interface` 定义策略/观察者/工厂契约并实现多态
- 实现工厂模式统一创建逻辑并用泛型工厂支持动态注册
- 实现观察者模式并返回取消订阅函数实现资源释放
- 实现策略模式在运行时切换算法并保持类型安全
- 实现适配器模式统一新旧接口并复用旧实现
- 实现单例模式（私有构造器 + 静态工厂或装饰器）
- 避免接口定义过度抽象导致实现类膨胀

## 1. 工厂模式

### Python 实现

```python
from abc import ABC, abstractmethod
from typing import Dict, Type

class Animal(ABC):
    @abstractmethod
    def speak(self) -> str:
        pass

class Dog(Animal):
    def speak(self) -> str:
        return "Woof!"

class Cat(Animal):
    def speak(self) -> str:
        return "Meow!"

class AnimalFactory:
    _animals: Dict[str, Type[Animal]] = {
        "dog": Dog,
        "cat": Cat
    }

    @classmethod
    def create(cls, animal_type: str) -> Animal:
        if animal_type not in cls._animals:
            raise ValueError(f"Unknown animal type: {animal_type}")
        return cls._animals[animal_type]()

# 使用
dog = AnimalFactory.create("dog")
print(dog.speak())  # Woof!
```

短对照：工厂方法核心结构对比。

```python
class AnimalFactory:
    _animals = {"dog": Dog, "cat": Cat}
    @classmethod
    def create(cls, t): return cls._animals[t]()
```

```typescript

}
```

### TypeScript 实现

```typescript
interface Animal {
  speak(): string;
}

class Dog implements Animal {
  speak(): string {
    return "Woof!";
  }
}

class Cat implements Animal {
  speak(): string {
    return "Meow!";
  }
}

type AnimalType = "dog" | "cat";

class AnimalFactory {
  private static animals: Record<AnimalType, new () => Animal> = {
    dog: Dog,
    cat: Cat,
  };

  static create(type: AnimalType): Animal {
    const AnimalClass = this.animals[type];
    if (!AnimalClass) {
      throw new Error(`Unknown animal type: ${type}`);
    }
    return new AnimalClass();
  }
}

// 使用
const dog = AnimalFactory.create("dog");
console.log(dog.speak()); // Woof!

// 泛型工厂
class GenericFactory<T> {
  private registry = new Map<string, new () => T>();

  register(type: string, constructor: new () => T): void {
    this.registry.set(type, constructor);
  }

  create(type: string): T {
    const Constructor = this.registry.get(type);
    if (!Constructor) {
      throw new Error(`Type ${type} not registered`);
    }
    return new Constructor();
  }
}
```

短对照：泛型工厂注册与创建。

```python
class GenericFactory:
    def __init__(self): self._reg = {}
    def register(self, t, c): self._reg[t] = c
    def create(self, t): return self._reg[t]()
```

```typescript
class GenericFactory<T> {
  private reg = new Map<string, new()=>T>();
  register(t: string, c: new()=>T) { this.reg.set(t, c); }
  create(t: string) { return new this.reg.get(t)!(); }
}
```

## 2. 观察者模式

### Python 实现

```python
from typing import List, Callable, Any

class Event:
    def __init__(self):
        self._listeners: List[Callable[..., None]] = []

    def subscribe(self, listener: Callable[..., None]) -> None:
        self._listeners.append(listener)

    def unsubscribe(self, listener: Callable[..., None]) -> None:
        self._listeners.remove(listener)

    def emit(self, *args, **kwargs) -> None:
        for listener in self._listeners:
            listener(*args, **kwargs)

class UserService:
    def __init__(self):
        self.user_created = Event()
        self.user_deleted = Event()

    def create_user(self, name: str) -> None:
        print(f"Creating user: {name}")
        self.user_created.emit(name)

    def delete_user(self, name: str) -> None:
        print(f"Deleting user: {name}")
        self.user_deleted.emit(name)

# 使用
service = UserService()
service.user_created.subscribe(lambda name: print(f"Notification: {name} created"))
service.create_user("Alice")
```

短对照：事件发射与订阅核心结构。

```python
class Event:
    def __init__(self): self._ls = []
    def sub(self, fn): self._ls.append(fn)
    def emit(self, *a): [f(*a) for f in self._ls]
```

```typescript
class EventEmitter<T> {
  private ls: ((d:T)=>void)[] = [];
  sub(fn: (d:T)=>void) { this.ls.push(fn); return ()=>{...}; }
  emit(d: T) { this.ls.forEach(f=>f(d)); }
}
```

### TypeScript 实现

```typescript
type EventCallback<T> = (data: T) => void;

class EventEmitter<T> {
  private listeners: EventCallback<T>[] = [];

  subscribe(listener: EventCallback<T>): () => void {
    this.listeners.push(listener);
    // 返回取消订阅函数
    return () => {
      const index = this.listeners.indexOf(listener);
      if (index > -1) {
        this.listeners.splice(index, 1);
      }
    };
  }

  emit(data: T): void {
    this.listeners.forEach((listener) => listener(data));
  }
}

interface UserEvent {
  name: string;
  timestamp: Date;
}

class UserService {
  readonly userCreated = new EventEmitter<UserEvent>();
  readonly userDeleted = new EventEmitter<UserEvent>();

  createUser(name: string): void {
    console.log(`Creating user: ${name}`);
    this.userCreated.emit({ name, timestamp: new Date() });
  }

  deleteUser(name: string): void {
    console.log(`Deleting user: ${name}`);
    this.userDeleted.emit({ name, timestamp: new Date() });
  }
}

// 使用
const service = new UserService();
const unsubscribe = service.userCreated.subscribe(({ name }) =>
  console.log(`Notification: ${name} created`)
);

service.createUser("Alice");
unsubscribe(); // 取消订阅
```

:::tip
订阅返回取消函数是 TypeScript 常见模式（如 RxJS、React hooks）。调用方只需保存返回值，销毁时调用即可，无需持有原始监听器引用。
:::

错误版订阅不返回取消函数导致无法清理，正确版返回取消函数：

```typescript
// ❌ 错误：subscribe 返回 void，无法取消订阅，内存泄漏风险
class BadEmitter<T> {
  private ls: ((d:T)=>void)[] = [];
  sub(fn: (d:T)=>void) { this.ls.push(fn); }
  emit(d: T) { this.ls.forEach(f=>f(d)); }
}

// ✅ 正确：返回取消函数，支持及时清理
class GoodEmitter<T> {
  private ls: ((d:T)=>void)[] = [];
  sub(fn: (d:T)=>void) {
    this.ls.push(fn);
    return () => { const i = this.ls.indexOf(fn); if (i>-1) this.ls.splice(i,1); };
  }
  emit(d: T) { this.ls.forEach(f=>f(d)); }
}
```

## 3. 策略模式

### Python 实现

```python
from abc import ABC, abstractmethod
from typing import List

class SortStrategy(ABC):
    @abstractmethod
    def sort(self, data: List[int]) -> List[int]:
        pass

class QuickSort(SortStrategy):
    def sort(self, data: List[int]) -> List[int]:
        if len(data) <= 1:
            return data
        pivot = data[len(data) // 2]
        left = [x for x in data if x < pivot]
        middle = [x for x in data if x == pivot]
        right = [x for x in data if x > pivot]
        return self.sort(left) + middle + self.sort(right)

class MergeSort(SortStrategy):
    def sort(self, data: List[int]) -> List[int]:
        if len(data) <= 1:
            return data
        mid = len(data) // 2
        left = self.sort(data[:mid])
        right = self.sort(data[mid:])
        return self._merge(left, right)

    def _merge(self, left: List[int], right: List[int]) -> List[int]:
        result = []
        i = j = 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                result.append(left[i])
                i += 1
            else:
                result.append(right[j])
                j += 1
        result.extend(left[i:])
        result.extend(right[j:])
        return result

class Sorter:
    def __init__(self, strategy: SortStrategy):
        self._strategy = strategy

    def sort(self, data: List[int]) -> List[int]:
        return self._strategy.sort(data)

    def set_strategy(self, strategy: SortStrategy) -> None:
        self._strategy = strategy

# 使用
sorter = Sorter(QuickSort())
result = sorter.sort([3, 1, 4, 1, 5])
print(result)  # [1, 1, 3, 4, 5]

sorter.set_strategy(MergeSort())
result = sorter.sort([3, 1, 4, 1, 5])
print(result)  # [1, 1, 3, 4, 5]
```

### TypeScript 实现

```typescript
interface SortStrategy {
  sort(data: number[]): number[];
}

class QuickSort implements SortStrategy {
  sort(data: number[]): number[] {
    if (data.length <= 1) return data;

    const pivot = data[Math.floor(data.length / 2)];
    const left = data.filter((x) => x < pivot);
    const middle = data.filter((x) => x === pivot);
    const right = data.filter((x) => x > pivot);

    return [...this.sort(left), ...middle, ...this.sort(right)];
  }
}

class MergeSort implements SortStrategy {
  sort(data: number[]): number[] {
    if (data.length <= 1) return data;

    const mid = Math.floor(data.length / 2);
    const left = this.sort(data.slice(0, mid));
    const right = this.sort(data.slice(mid));

    return this.merge(left, right);
  }

  private merge(left: number[], right: number[]): number[] {
    const result: number[] = [];
    let i = 0,
      j = 0;

    while (i < left.length && j < right.length) {
      if (left[i] <= right[j]) {
        result.push(left[i]);
        i++;
      } else {
        result.push(right[j]);
        j++;
      }
    }

    return [...result, ...left.slice(i), ...right.slice(j)];
  }
}

class Sorter {
  private strategy: SortStrategy;

  constructor(strategy: SortStrategy) {
    this.strategy = strategy;
  }

  sort(data: number[]): number[] {
    return this.strategy.sort(data);
  }

  setStrategy(strategy: SortStrategy): void {
    this.strategy = strategy;
  }
}

// 使用
const sorter = new Sorter(new QuickSort());
console.log(sorter.sort([3, 1, 4, 1, 5])); // [1, 1, 3, 4, 5]

sorter.setStrategy(new MergeSort());
console.log(sorter.sort([3, 1, 4, 1, 5])); // [1, 1, 3, 4, 5]
```

  ts={`class Sorter {

}`}

:::note
策略模式的关键是 `interface` 定义契约，`implements` 保证实现完整。TypeScript 编译期会强制所有策略实现 `sort` 方法，Python ABC 只在实例化时检查。
:::

错误版直接用条件分支硬编码算法，正确版抽象策略接口：

```typescript
// ❌ 错误：算法硬编码在 Sorter 中，违反开闭原则
class BadSorter {
  sort(data: number[], algo: "quick" | "merge"): number[] {
    if (algo === "quick") { /* quick sort logic */ return data; }
    else { /* merge sort logic */ return data; }
  }
}

// ✅ 正确：策略独立，新增算法不修改 Sorter
interface SortStrategy { sort(data: number[]): number[]; }
class QuickSort implements SortStrategy { sort(d) { ... } }
class MergeSort implements SortStrategy { sort(d) { ... } }
class GoodSorter {
  constructor(private s: SortStrategy) {}
  sort(d: number[]) { return this.s.sort(d); }
  setStrategy(s: SortStrategy) { this.s = s; }
}
```

## 4. 适配器模式

### Python 实现

```python
from abc import ABC, abstractmethod

class OldPrinter:
    def old_print(self, text: str) -> None:
        print(f"Old: {text}")

class NewPrinter:
    def print(self, text: str) -> None:
        print(f"New: {text}")

class PrinterAdapter:
    def __init__(self, printer: OldPrinter):
        self._printer = printer

    def print(self, text: str) -> None:
        self._printer.old_print(text)

# 使用
old_printer = OldPrinter()
adapter = PrinterAdapter(old_printer)
adapter.print("Hello")  # 统一使用 print 方法
```

短对照：适配器包装旧接口转新接口。

<CodeCompare
  py={`class PrinterAdapter:
    def __init__(self, p): self._p = p
    def print(self, t): self._p.old_print(t)`}
  ts={`class OldPrinterAdapter implements Printer {
  constructor(private p: OldPrinter) {}
  print(t: string) { this.p.oldPrint(t); }
}`}
/>

### TypeScript 实现

```typescript
interface Printer {
  print(text: string): void;
}

class OldPrinter {
  oldPrint(text: string): void {
    console.log(`Old: ${text}`);
  }
}

class NewPrinter implements Printer {
  print(text: string): void {
    console.log(`New: ${text}`);
  }
}

class OldPrinterAdapter implements Printer {
  private printer: OldPrinter;

  constructor(printer: OldPrinter) {
    this.printer = printer;
  }

  print(text: string): void {
    this.printer.oldPrint(text);
  }
}

// 使用
function printDocument(printer: Printer, text: string): void {
  printer.print(text);
}

const oldPrinter = new OldPrinter();
const adapter = new OldPrinterAdapter(oldPrinter);
printDocument(adapter, "Hello"); // 统一接口

const newPrinter = new NewPrinter();
printDocument(newPrinter, "World"); // 同样的接口
```

:::tip
适配器模式的核心是 `implements` 目标接口，构造函数接收被适配对象。TypeScript 编译期会检查适配器是否实现了 `Printer` 的所有方法，防止漏写。
:::

错误版适配器未实现接口导致类型不匹配，正确版显式 implements：

```typescript
// ❌ 错误：未 implements Printer，printDocument 接受时类型报错
class BadAdapter {
  constructor(private p: OldPrinter) {}
  print(t: string) { this.p.oldPrint(t); }
}

// ✅ 正确：显式实现接口，编译期保证契约完整
class GoodAdapter implements Printer {
  constructor(private p: OldPrinter) {}
  print(t: string) { this.p.oldPrint(t); }
}
```

## 差异与优势

| 特性       | Python         | TypeScript         |
| ---------- | -------------- | ------------------ |
| 接口定义   | ABC/Protocol   | interface          |
| 类型检查   | 运行时         | 编译时             |
| 泛型支持   | 有限           | 完整               |
| 抽象类     | ABC            | abstract class     |
| 实现灵活性 | 高（鸭子类型） | 中等（需实现接口） |

## 练习

将以下 Python 单例模式转换为 TypeScript：

```python
class Singleton:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        self.value = 0

# 测试
a = Singleton()
b = Singleton()
print(a is b)  # True
```

<details>
<summary>答案</summary>

```typescript
class Singleton {
  private static instance: Singleton;
  value: number = 0;

  private constructor() {}

  static getInstance(): Singleton {
    if (!Singleton.instance) {
      Singleton.instance = new Singleton();
    }
    return Singleton.instance;
  }
}

// 测试
const a = Singleton.getInstance();
const b = Singleton.getInstance();
console.log(a === b); // true

// 或者使用装饰器
function singleton<T extends new (...args: any[]) => {}>(constructor: T) {
  let instance: InstanceType<T> | undefined;
  return class extends constructor {
    constructor(...args: any[]) {
      if (instance) {
        return instance;
      }
      super(...args);
      // 派生类实例与基类实例类型结构兼容，TS 无法自动推导，需显式断言
      instance = this as InstanceType<T>;
    }
  };
}

@singleton
class Config {
  apiUrl = "https://api.example.com";
}

const config1 = new Config();
const config2 = new Config();
console.log(config1 === config2); // true
```

</details>

:::note
单例模式的关键点：私有构造器防止外部 `new`，静态实例字段保存唯一实例，静态方法控制创建时机。装饰器版本通过返回包装类实现，更优雅但需启用实验性装饰器。
:::

错误版构造器公开允许直接 new，正确版私有构造器强制走 getInstance：

```typescript
// ❌ 错误：构造器公开，外部可直接 new Singleton() 破坏单例
class BadSingleton {
  static instance: BadSingleton;
  static getInstance() { ... }
  constructor() {} // 公开构造器
}
const b1 = new BadSingleton(); // 绕过 getInstance
const b2 = BadSingleton.getInstance();
console.log(b1 === b2); // false！

// ✅ 正确：私有构造器，只能通过 getInstance 获取
class GoodSingleton {
  private static instance: GoodSingleton;
  private constructor() {} // 私有构造器
  static getInstance() {
    if (!GoodSingleton.instance) GoodSingleton.instance = new GoodSingleton();
    return GoodSingleton.instance;
  }
}
const g1 = GoodSingleton.getInstance();
const g2 = GoodSingleton.getInstance();
console.log(g1 === g2); // true
```

## 面试追问

1. **什么时候使用工厂模式而不是直接实例化？**
   - 答：当创建逻辑复杂、需要根据不同条件创建不同类型、或者需要将创建和使用分离时。

2. **观察者模式和发布-订阅模式有什么区别？**
   - 答：观察者模式中观察者和被观察者直接通信；发布-订阅模式通过消息中心解耦。

3. **策略模式的优势是什么？**
   - 答：避免大量条件语句，易于扩展新算法，算法可以独立变化。
